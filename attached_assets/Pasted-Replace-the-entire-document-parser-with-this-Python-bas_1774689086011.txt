import { execFileSync } from 'child_process'
import * as path from 'path'
import * as fs from 'fs'
import * as os from 'os'

export async function detectAndParse(
  buffer: Buffer,
  filename: string,
  mimeType: string
) {
  // Write buffer to temp file
  const tmpDir = os.tmpdir()
  const tmpFile = path.join(tmpDir, 
    `upload_${Date.now()}_${filename}`)
  
  try {
    fs.writeFileSync(tmpFile, buffer)
    
    const scriptPath = path.join(
      __dirname, 'document_reader.py'
    )
    
    console.log(`📄 Processing: ${filename}`)
    
    const output = execFileSync('python3', [scriptPath, tmpFile], {
      timeout: 120000,  // 2 min timeout
      maxBuffer: 10 * 1024 * 1024  // 10MB output buffer
    })
    
    const result = JSON.parse(output.toString())
    
    console.log(`✅ Parsed with ${result.method}:`, 
      result.total_found, 'transactions found')
    
    return {
      success: true,
      transactions: result.transactions || [],
      totalFound: result.total_found || 0,
      method: result.method,
      currency: result.currency,
      parseMethod: result.method
    }
    
  } catch(e: any) {
    console.error('❌ Parser error:', e.message)
    
    // Check if it's a timeout
    if (e.message?.includes('timeout') || 
        e.signal === 'SIGTERM') {
      return {
        success: false,
        locked: true,
        transactions: [],
        totalFound: 0,
        message: 'timeout'
      }
    }
    
    return {
      success: false,
      locked: true,
      transactions: [],
      totalFound: 0,
      message: 'parse_failed'
    }
    
  } finally {
    // Always clean up temp file
    try { fs.unlinkSync(tmpFile) } catch {}
  }
}
