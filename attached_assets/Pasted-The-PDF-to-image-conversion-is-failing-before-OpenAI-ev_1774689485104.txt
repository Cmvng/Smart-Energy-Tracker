async function pdfToImages(buffer: Buffer): 
  Promise<Buffer[]> {
  
  const os = require('os')
  const path = require('path')
  const fs = require('fs')
  const { execSync, execFileSync } = require('child_process')
  
  const tmpDir = fs.mkdtempSync(
    path.join(os.tmpdir(), 'pdf_')
  )
  const pdfPath = path.join(tmpDir, 'input.pdf')
  const outPrefix = path.join(tmpDir, 'page')
  
  try {
    fs.writeFileSync(pdfPath, buffer)
    console.log('PDF written:', buffer.length, 'bytes')
    console.log('PDF path:', pdfPath)
    
    // Try pdftoppm first
    try {
      execFileSync('pdftoppm', [
        '-r', '150',
        '-png',
        '-f', '1',
        '-l', '10',
        pdfPath,
        outPrefix
      ], { timeout: 60000 })
      console.log('pdftoppm succeeded')
    } catch(e1) {
      console.log('pdftoppm failed:', e1.message)
      
      // Try ghostscript
      try {
        execFileSync('gs', [
          '-dNOPAUSE',
          '-dBATCH', 
          '-dSAFER',
          '-sDEVICE=png16m',
          '-r150',
          `-sOutputFile=${outPrefix}-%d.png`,
          pdfPath
        ], { timeout: 60000 })
        console.log('ghostscript succeeded')
      } catch(e2) {
        console.log('ghostscript failed:', e2.message)
        
        // Try python pdf2image as last resort
        const pyScript = `
import sys, json
from pdf2image import convert_from_path
pages = convert_from_path('${pdfPath}', dpi=150, 
  first_page=1, last_page=10)
for i, page in enumerate(pages):
    page.save('${outPrefix}-{}.png'.format(i+1))
print(len(pages))
`
        const pyResult = execSync(
          `python3 -c "${pyScript.replace(/"/g, '\\"').replace(/\n/g, ' ')}"`
        ).toString()
        console.log('pdf2image succeeded, pages:', pyResult.trim())
      }
    }
    
    // Read all generated PNG files
    const files = fs.readdirSync(tmpDir)
      .filter((f: string) => f.endsWith('.png'))
      .sort()
    
    console.log('PNG files generated:', files.length)
    console.log('Files:', files)
    
    if (files.length === 0) {
      console.log('ERROR: No PNG files generated')
      console.log('Contents of tmpDir:', 
        fs.readdirSync(tmpDir))
      return []
    }
    
    const images = files.map((f: string) => 
      fs.readFileSync(path.join(tmpDir, f))
    )
    
    console.log('Image sizes:', 
      images.map((img: Buffer) => img.length + ' bytes'))
    
    return images
    
  } catch(e: any) {
    console.error('pdfToImages FATAL error:', e.message)
    console.error('Stack:', e.stack)
    return []
  } finally {
    // Cleanup
    try {
      fs.rmSync(tmpDir, { recursive: true })
    } catch {}
  }
}
