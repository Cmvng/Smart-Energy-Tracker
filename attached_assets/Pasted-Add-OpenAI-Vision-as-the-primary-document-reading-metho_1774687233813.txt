MAIN PROCESSING FUNCTION:
async function detectAndParse(
  buffer: Buffer, 
  filename: string, 
  mimeType: string
) {
  // CSV - always fast and free
  if (filename.endsWith('.csv') || 
      mimeType.includes('csv')) {
    return universalCSVParser(buffer)
  }
  
  // Image uploaded directly
  if (mimeType.includes('image')) {
    if (!process.env.OPENAI_API_KEY) {
      return { 
        error: true, 
        message: 'AI scanning not configured' 
      }
    }
    return readWithAI([buffer], filename)
  }
  
  // PDF processing
  if (filename.endsWith('.pdf') || 
      mimeType.includes('pdf')) {
    
    // Step 1: Try free text extraction first
    try {
      const textResult = await parsePDF(buffer)
      if (textResult.transactions?.length >= 5) {
        console.log('✅ PDF parsed with text extraction')
        return textResult
      }
    } catch(e) {
      console.log('Text extraction failed:', e.message)
    }
    
    // Step 2: Convert to images and use AI Vision
    if (!process.env.OPENAI_API_KEY) {
      // No AI key - show helpful message
      return {
        locked: true,
        transactions: [],
        totalFound: 0,
        message: 'This PDF needs AI scanning. CSV format works instantly.'
      }
    }
    
    console.log('🔍 Using AI Vision for PDF...')
    const images = await pdfToImages(buffer)
    
    if (images.length === 0) {
      return {
        locked: true,
        transactions: [],
        totalFound: 0
      }
    }
    
    return await readWithAI(images, filename)
  }
  
  return { error: true, message: 'Unsupported file type' }
}
