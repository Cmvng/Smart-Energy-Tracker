async function getUserByChatId(chatId) {
  const result = await db.query(
    'SELECT * FROM users WHERE telegram_chat_id = $1',
    [String(chatId)]
  )
  return result.rows[0] || null
}

async function getUserAccount(userId) {
  const result = await db.query(
    'SELECT * FROM accounts WHERE user_id = $1 LIMIT 1',
    [userId]
  )
  return result.rows[0] || null
}

async function getTodaySummary(accountId) {
  const todayStart = new Date()
  todayStart.setHours(0, 0, 0, 0)
  const result = await db.query(`
    SELECT 
      COALESCE(SUM(CASE WHEN type='income' 
        THEN amount_usd ELSE 0 END), 0) as income,
      COALESCE(SUM(CASE WHEN type='expense' 
        THEN amount_usd ELSE 0 END), 0) as expenses,
      COUNT(*) as count
    FROM transactions 
    WHERE account_id = $1 
    AND deleted_at IS NULL
    AND transacted_at >= $2
  `, [accountId, todayStart])
  const row = result.rows[0]
  const income = parseFloat(row.income)
  const expenses = parseFloat(row.expenses)
  return { income, expenses, net: income - expenses, 
           count: parseInt(row.count) }
}

async function getWeekSummary(accountId) {
  const weekStart = new Date()
  weekStart.setDate(weekStart.getDate() - weekStart.getDay() + 1)
  weekStart.setHours(0, 0, 0, 0)
  const result = await db.query(`
    SELECT 
      COALESCE(SUM(CASE WHEN type='income' 
        THEN amount_usd ELSE 0 END), 0) as income,
      COALESCE(SUM(CASE WHEN type='expense' 
        THEN amount_usd ELSE 0 END), 0) as expenses,
      COUNT(*) as count
    FROM transactions 
    WHERE account_id = $1 
    AND deleted_at IS NULL
    AND transacted_at >= $2
  `, [accountId, weekStart])
  const row = result.rows[0]
  const income = parseFloat(row.income)
  const expenses = parseFloat(row.expenses)
  return { income, expenses, net: income - expenses,
           count: parseInt(row.count) }
}

function formatMoney(n) {
  return '$' + Math.abs(n).toFixed(2).replace(
    /\B(?=(\d{3})+(?!\d))/g, ','
  )
}

function netEmoji(net) {
  if (net > 0) return '🟢 Profit'
  if (net < 0) return '🔴 Loss'
  return '🟡 Breakeven'
}

async (msg, match) => {
  const chatId = msg.chat.id
  const type = match[1]
  const amountRaw = parseFloat(match[2])
  const currencyRaw = match[3] ? match[3].toUpperCase() : null
  const note = match[4] || ''
  
  if (isNaN(amountRaw) || amountRaw <= 0) {
    bot.sendMessage(chatId,
      `🤔 Invalid amount. Try: /add ${type} 5000 NGN groceries`)
    return
  }
  
  try {
    const user = await getUserByChatId(chatId)
    if (!user) { bot.sendMessage(chatId, NOT_LINKED); return }

    if (!account) {
      bot.sendMessage(chatId,
        '😅 No account found. Please open the app first.')
      return
    }
    
    // Get currency - use provided or fall back to user home
    const currency = currencyRaw || user.home_currency || 'USD'
    const rateResult = await db.query(
      'SELECT rate_to_usd, name FROM currencies WHERE code = $1',
      [currency]
    )
    const rate = rateResult.rows.length > 0 
      ? parseFloat(rateResult.rows[0].rate_to_usd) 
      : 1.0
    const usedFallback = rateResult.rows.length === 0
    const amount_usd = amountRaw * rate

      INSERT INTO transactions

    try {
      const user = await getUserByChatId(chatId)
      if (!user) { bot.sendMessage(chatId, NOT_LINKED); return }

      await db.query(`
        INSERT INTO telegram_sessions

        ON CONFLICT (user_id) 
        DO UPDATE SET state='awaiting_type', 
          data=$2, updated_at=NOW()
      `, [user.id, JSON.stringify({ amount })])
      
      bot.sendMessage(chatId,
        `Got ${user.home_currency || 'USD'} ${amount.toLocaleString()}! What was this? 👇`,

            inline_keyboard: [[
              { text: '💰 Income', callback_data: 'type_income' },

            ]]
          }
        }
      )
    } catch(e) {

  }
  
  // Unknown message that is not a command or number
  if (!text.startsWith('/')) {

CALLBACK QUERY HANDLER (inline button taps):
bot.on('callback_query', async (query) => {
  const chatId = query.message.chat.id
  const data = query.data
  
  try {
    const user = await getUserByChatId(chatId)
    if (!user) { 
      bot.answerCallbackQuery(query.id, 
        { text: 'Please link your account first' })
      return 
    }
    
    if (data === 'type_income' || data === 'type_expense') {
      const type = data === 'type_income' ? 'income' : 'expense'
      
      // Get amount from session
      const sessionResult = await db.query(
        `SELECT data FROM telegram_sessions 
         WHERE user_id = $1 AND state = 'awaiting_type'`,
        [user.id]
      )
      
      if (sessionResult.rows.length === 0) {
        bot.answerCallbackQuery(query.id, 
          { text: 'Session expired, please try again' })
        return
      }
      
      const { amount } = sessionResult.rows[0].data
      const currency = user.home_currency || 'USD'
      
      const rateResult = await db.query(
        'SELECT rate_to_usd FROM currencies WHERE code = $1',
        [currency]
      )
      const rate = rateResult.rows.length > 0 
        ? parseFloat(rateResult.rows[0].rate_to_usd) : 1.0
      const amount_usd = amount * rate
      
      const account = await getUserAccount(user.id)
      
      await db.query(`
        INSERT INTO transactions
          (account_id, type, amount_original, currency_code,
           amount_usd, fx_rate_used, notes, transacted_at)
        VALUES ($1,$2,$3,$4,$5,$6,'',NOW())
      `, [account.id, type, amount, currency, amount_usd, rate])
      
      // Clear session
      await db.query(
        'DELETE FROM telegram_sessions WHERE user_id = $1',
        [user.id]
      )
      
      const s = await getTodaySummary(account.id)
      
      bot.answerCallbackQuery(query.id, { text: '✅ Saved!' })
      bot.editMessageText(
`✅ ${type === 'income' ? 'Income' : 'Expense'} saved!
${currency} ${amount.toLocaleString()} = ${formatMoney(amount_usd)} USD

📊 Today: Income ${formatMoney(s.income)} · Expenses ${formatMoney(s.expenses)}
${netEmoji(s.net)}: ${s.net >= 0 ? '+' : '-'}${formatMoney(s.net)}`,
        { 
          chat_id: chatId, 
          message_id: query.message.message_id 
        }
      )
    }
  } catch(e) {
    console.error('callback error:', e)
    bot.answerCallbackQuery(query.id, 
      { text: '😅 Something went wrong' })
  }
})

import cron from 'node-cron'

// Morning: 8AM UTC daily
cron.schedule('0 8 * * *', async () => {
  try {
    const users = await db.query(`
      SELECT * FROM users 
      WHERE telegram_chat_id IS NOT NULL 
      AND notification_frequency != 'off'
    `)
    const dayOfWeek = new Date().getDay() // 0=Sun,1=Mon...
    
    for (const user of users.rows) {
      const freq = user.notification_frequency
      // Check if should send today
      const shouldSend = 
        freq === 'daily' ||
        (freq === 'every3days' && [1,3,5].includes(dayOfWeek)) ||
        (freq === 'weekly' && dayOfWeek === 0)
      
      if (!shouldSend) continue
      
      try {
        const account = await getUserAccount(user.id)
        if (!account) continue
        
        // Get yesterday summary
        const yesterday = new Date()
        yesterday.setDate(yesterday.getDate() - 1)
        yesterday.setHours(0,0,0,0)
        const yesterdayEnd = new Date(yesterday)
        yesterdayEnd.setHours(23,59,59,999)
        
        const yResult = await db.query(`
          SELECT 
            COALESCE(SUM(CASE WHEN type='income' 
              THEN amount_usd ELSE 0 END),0) as income,
            COALESCE(SUM(CASE WHEN type='expense' 
              THEN amount_usd ELSE 0 END),0) as expenses
          FROM transactions
          WHERE account_id=$1 
          AND deleted_at IS NULL
          AND transacted_at BETWEEN $2 AND $3
        `, [account.id, yesterday, yesterdayEnd])
        
        const yi = parseFloat(yResult.rows[0].income)
        const ye = parseFloat(yResult.rows[0].expenses)
