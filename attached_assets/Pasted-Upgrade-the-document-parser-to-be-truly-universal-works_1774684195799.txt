function detectTransactionType(
  description: string,
  amount: number,
  isNegative: boolean,
  hasDebitCol: boolean,
  hasCreditCol: boolean,
  debitValue: number | null,
  creditValue: number | null
): 'income' | 'expense' {
  
  const desc = description.toLowerCase()
  
  // Method 1: Separate debit/credit columns
  if (hasDebitCol && hasCreditCol) {
    if (creditValue && creditValue > 0) return 'income'
    if (debitValue && debitValue > 0) return 'expense'
  }
  
  // Method 2: Check skip keywords first
  for (const skip of SKIP_KEYWORDS) {
    if (desc.includes(skip)) return null // skip this row
  }
  
  // Method 3: Check income keywords
  for (const keyword of INCOME_KEYWORDS) {
    if (desc.includes(keyword)) return 'income'
  }
  
  // Method 4: Check expense keywords  
  for (const keyword of EXPENSE_KEYWORDS) {
    if (desc.includes(keyword)) return 'expense'
  }
  
  // Method 5: Amount sign
  if (isNegative) return 'expense'
  if (amount > 0 && !isNegative) return 'income'
  
  // Method 6: Default to expense if unsure
  // (most transactions are expenses)
  return 'expense'
}

function parseAmount(str: string): {
  value: number, 
  isNegative: boolean,
  currency: string | null
} {
  if (!str || str.trim() === '' || 
      str.trim() === '-' || str.trim() === '0') {
    return { value: 0, isNegative: false, currency: null }
  }
  
  let s = str.trim()
  let isNegative = false
  let currency = null
  
  // Detect negative: minus sign or parentheses
  if (s.startsWith('-') || s.startsWith('(') || 
      s.endsWith(')') || s.includes('DR') || 
      s.includes(' dr')) {
    isNegative = true
  }
  
  // Extract currency symbols and codes
  const currencyMap = {
    '$': 'USD', '£': 'GBP', '€': 'EUR',
    '₦': 'NGN', '₵': 'GHS', 'GHc': 'GHS',
    '₹': 'INR', '¥': 'JPY', '¥': 'CNY',
    'R': 'ZAR', 'KSh': 'KES', 'USh': 'UGX',
    'TSh': 'TZS', 'RWF': 'RWF', 'XOF': 'XOF',
    'XAF': 'XAF', 'ETB': 'ETB', 'GHS': 'GHS',
    'NGN': 'NGN', 'USD': 'USD', 'GBP': 'GBP',
    'EUR': 'EUR', 'CAD': 'CAD', 'AUD': 'AUD',
    'ZAR': 'ZAR', 'KES': 'KES', 'INR': 'INR',
    'JPY': 'JPY', 'CNY': 'CNY', 'BRL': 'BRL',
    'MXN': 'MXN', 'AED': 'AED', 'SAR': 'SAR',
    'CHF': 'CHF', 'SEK': 'SEK', 'NOK': 'NOK',
    'DKK': 'DKK', 'PLN': 'PLN', 'CZK': 'CZK',
    'HUF': 'HUF', 'MYR': 'MYR', 'SGD': 'SGD',
    'HKD': 'HKD', 'THB': 'THB', 'IDR': 'IDR',
    'PHP': 'PHP', 'PKR': 'PKR', 'BDT': 'BDT',
    'EGP': 'EGP', 'MAD': 'MAD', 'TND': 'TND'
  }
  
  for (const [symbol, code] of Object.entries(currencyMap)) {
    if (s.includes(symbol)) {
      currency = code
      s = s.replace(symbol, '')
      break
    }
  }
  
  // Remove parentheses, DR/CR labels
  s = s.replace(/[()DR]/gi, '')
       .replace(/\s+/g, '')
  
  // Handle European format: 1.234,56 → 1234.56
  // Detect: if comma comes after dot → European format
  if (/\d+\.\d{3},\d{2}/.test(s)) {
    s = s.replace(/\./g, '').replace(',', '.')
  }
  // Handle: 1,234.56 → standard
  else {
    s = s.replace(/,/g, '')
  }
  
  const value = Math.abs(parseFloat(s))
  
  if (isNaN(value) || value === 0) {
    return { value: 0, isNegative: false, currency: null }
  }
  
  return { value, isNegative, currency }
}

function parseDate(str: string): string | null {
  if (!str || str.trim() === '') return null
  
  const s = str.trim()
  
  // Try these patterns in order:
  const patterns = [
    // DD/MM/YYYY or DD-MM-YYYY
    { regex: /(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{4})/, 
      parse: (m) => `${m[3]}-${m[2].padStart(2,'0')}-${m[1].padStart(2,'0')}` },
    // YYYY-MM-DD or YYYY/MM/DD
    { regex: /(\d{4})[\/\-\.](\d{1,2})[\/\-\.](\d{1,2})/, 
      parse: (m) => `${m[1]}-${m[2].padStart(2,'0')}-${m[3].padStart(2,'0')}` },
    // DD MMM YYYY (01 Mar 2026)
    { regex: /(\d{1,2})\s+(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\w*\s+(\d{4})/i,
      parse: (m) => {
        const months = {jan:'01',feb:'02',mar:'03',apr:'04',may:'05',jun:'06',
                       jul:'07',aug:'08',sep:'09',oct:'10',nov:'11',dec:'12'}
        return `${m[3]}-${months[m[2].toLowerCase().slice(0,3)]}-${m[1].padStart(2,'0')}`
      }
    },
    // MMM DD YYYY (Mar 01 2026)
    { regex: /(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\w*\s+(\d{1,2}),?\s+(\d{4})/i,
      parse: (m) => {
        const months = {jan:'01',feb:'02',mar:'03',apr:'04',may:'05',jun:'06',
                       jul:'07',aug:'08',sep:'09',oct:'10',nov:'11',dec:'12'}
        return `${m[3]}-${months[m[1].toLowerCase().slice(0,3)]}-${m[2].padStart(2,'0')}`
      }
    },
    // DDMMMYYYY (01Mar2026)
    { regex: /(\d{2})(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)(\d{4})/i,
      parse: (m) => {
        const months = {jan:'01',feb:'02',mar:'03',apr:'04',may:'05',jun:'06',
                       jul:'07',aug:'08',sep:'09',oct:'10',nov:'11',dec:'12'}
        return `${m[3]}-${months[m[2].toLowerCase().slice(0,3)]}-${m[1]}`
      }
    },
    // DD/MM/YY
    { regex: /(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{2})$/,
      parse: (m) => {
        const year = parseInt(m[3]) > 50 
          ? '19'+m[3] : '20'+m[3]
        return `${year}-${m[2].padStart(2,'0')}-${m[1].padStart(2,'0')}`
      }
    },
    // MM/DD/YYYY (US format — only if month ≤ 12 AND day > 12)
    { regex: /(\d{1,2})[\/](\d{1,2})[\/](\d{4})/,
      parse: (m) => {
        // Ambiguous — default to DD/MM/YYYY unless
        // first number > 12 (must be a day in MM/DD)
        // or second number > 12 (must be a day in DD/MM)
        const a = parseInt(m[1]), b = parseInt(m[2])
        if (a > 12) {
          // first is day, second is month (DD/MM)
          return `${m[3]}-${m[2].padStart(2,'0')}-${m[1].padStart(2,'0')}`
        } else if (b > 12) {
          // second is day, first is month (MM/DD)  
          return `${m[3]}-${m[1].padStart(2,'0')}-${m[2].padStart(2,'0')}`
        } else {
          // Truly ambiguous — default DD/MM/YYYY
          return `${m[3]}-${m[2].padStart(2,'0')}-${m[1].padStart(2,'0')}`
        }
      }
    }
  ]
  
  for (const { regex, parse } of patterns) {
    const match = s.match(regex)
    if (match) {
      const result = parse(match)
      // Validate the result
      const d = new Date(result)
      if (!isNaN(d.getTime()) && 
          d.getFullYear() >= 2000 && 
          d.getFullYear() <= 2030) {
        return result
      }
    }
  }
  
  return null
}

function detectCurrency(text: string): string {
  const t = text
  
  // Count currency symbol occurrences
  const counts = {
    'NGN': (t.match(/₦|NGN/g) || []).length,
    'GHS': (t.match(/₵|GHc|GHS/g) || []).length,
    'KES': (t.match(/KSh|KES/g) || []).length,
    'ZAR': (t.match(/ZAR|\bR\b/g) || []).length,
    'GBP': (t.match(/£|GBP/g) || []).length,
    'EUR': (t.match(/€|EUR/g) || []).length,
    'USD': (t.match(/\$|USD/g) || []).length,
    'INR': (t.match(/₹|INR/g) || []).length,
    'UGX': (t.match(/USh|UGX/g) || []).length,
    'TZS': (t.match(/TSh|TZS/g) || []).length,
  }
  
  // Return the most frequently appearing currency
  const sorted = Object.entries(counts)
    .sort(([,a],[,b]) => b - a)
  
  if (sorted[0][1] > 0) return sorted[0][0]
  
  return 'USD' // fallback
}
