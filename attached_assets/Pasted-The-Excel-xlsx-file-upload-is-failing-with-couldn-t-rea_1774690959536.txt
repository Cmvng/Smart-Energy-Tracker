The Excel (.xlsx) file upload is failing with 
"couldn't read the column structure" because 
the xlsx parser is not connected to the 
document processing pipeline.

Fix this completely:

1. Install xlsx package if not installed:
   npm install xlsx

2. In server/document-parser.ts add this 
   function at the top:

function parseExcel(buffer: Buffer) {
  const XLSX = require('xlsx')
  
  try {
    const workbook = XLSX.read(buffer, { 
      type: 'buffer',
      cellDates: true,
      dateNF: 'dd/mm/yyyy'
    })
    
    // Use first sheet
    const sheetName = workbook.SheetNames[0]
    const sheet = workbook.Sheets[sheetName]
    
    // Convert to CSV string
    const csv = XLSX.utils.sheet_to_csv(sheet, {
      blankrows: false,
      skipHidden: true
    })
    
    console.log('Excel converted to CSV, length:', csv.length)
    console.log('First 300 chars:', csv.substring(0, 300))
    
    // Pass through existing CSV parser
    return universalCSVParser(Buffer.from(csv))
    
  } catch(e: any) {
    console.error('Excel parse error:', e.message)
    return { 
      error: true, 
      message: 'Could not read Excel file',
      transactions: [],
      totalFound: 0
    }
  }
}

const ext = filename.toLowerCase()

4. Also fix universalCSVParser to handle 
   the output from XLSX.utils.sheet_to_csv
   which sometimes produces slightly different
   formatting than raw bank CSV files.
   
   The main difference: Excel dates sometimes 
   export as numbers like 45900 instead of 
   01/03/2026.
   
   Add Excel date number conversion:
   
   function convertExcelDate(value: any): string | null {
     // If it's already a date string, parse normally
     const dateStr = parseDate(String(value))
     if (dateStr) return dateStr
     
     // If it's an Excel serial number (like 45900)
     const num = parseFloat(String(value))
     if (!isNaN(num) && num > 40000 && num < 60000) {
       // Convert Excel serial to JS date
       const date = new Date((num - 25569) * 86400 * 1000)
       if (!isNaN(date.getTime())) {
         return date.toISOString().split('T')[0]
       }
     }
     return null
   }
   
   Use convertExcelDate() instead of parseDate() 
   when processing rows from Excel conversion.
