Add admin middleware:
async function requireAdmin(req, res, next) {
  const user = await getUserFromJWT(req)
  if (!user || !user.is_admin) {
    return res.status(403).json({ error: 'Unauthorized' })
  }
  req.user = user
  next()
}

GET /api/admin/stats (requireAdmin):
Run these queries and return combined result:
{
  total_users: SELECT COUNT(*) FROM users,
  new_today: SELECT COUNT(*) FROM users 
    WHERE created_at >= CURRENT_DATE,
  new_this_week: SELECT COUNT(*) FROM users 
    WHERE created_at >= NOW() - INTERVAL '7 days',
  total_transactions: SELECT COUNT(*) FROM transactions 
    WHERE deleted_at IS NULL,
  transactions_today: SELECT COUNT(*) FROM transactions 
    WHERE deleted_at IS NULL 
    AND created_at >= CURRENT_DATE,
  active_today: SELECT COUNT(DISTINCT account_id) 
    FROM transactions 
    WHERE deleted_at IS NULL 
    AND created_at >= CURRENT_DATE,
  users_by_currency: SELECT home_currency, 
    COUNT(*) as count FROM users 
    GROUP BY home_currency 
    ORDER BY count DESC,
  recent_signups: SELECT 
    u.id, u.name, u.email, u.nickname,
    u.mode, u.home_currency, u.created_at,
    u.telegram_chat_id IS NOT NULL as telegram_connected,
    COUNT(t.id) as transaction_count
    FROM users u
    LEFT JOIN accounts a ON a.user_id = u.id
    LEFT JOIN transactions t ON t.account_id = a.id 
      AND t.deleted_at IS NULL
    GROUP BY u.id
    ORDER BY u.created_at DESC
    LIMIT 20
}

GET /api/admin/users (requireAdmin):
SELECT u.id, u.name, u.email, u.nickname,
  u.mode, u.home_currency, u.created_at,
  u.telegram_chat_id IS NOT NULL as telegram_connected,
  COUNT(t.id) as transaction_count
FROM users u
LEFT JOIN accounts a ON a.user_id = u.id
LEFT JOIN transactions t ON t.account_id = a.id 
  AND t.deleted_at IS NULL
WHERE ($1::text IS NULL OR 
  u.name ILIKE '%' || $1 || '%' OR 
  u.email ILIKE '%' || $1 || '%')
GROUP BY u.id
ORDER BY u.created_at DESC
($1 = req.query.search or null)

GET /api/admin/feedback (requireAdmin):
SELECT f.*, u.name as user_name, u.email as user_email
FROM feedback f
LEFT JOIN users u ON u.id = f.user_id
ORDER BY f.created_at DESC
