const textLength = data.text?.trim().length || 0
const pageCount = data.numpages || 1

if (textLength < 100) {
  return {
    success: false,
    locked: true,
    transactions: [],
    totalFound: 0,
    message: 'locked_pdf',
    pageCount: pageCount
  }
}
