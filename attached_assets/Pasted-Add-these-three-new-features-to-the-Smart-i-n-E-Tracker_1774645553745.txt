- ?timeframe=day|week|month|year (existing)

When from+to params present:
SELECT 
  COALESCE(SUM(CASE WHEN type='income' 
    THEN amount_usd ELSE 0 END),0) as income,
  COALESCE(SUM(CASE WHEN type='expense' 
    THEN amount_usd ELSE 0 END),0) as expenses,
  COUNT(*) as count
FROM transactions
WHERE account_id = [user account id]
AND deleted_at IS NULL
AND transacted_at >= [from date 00:00:00]
AND transacted_at <= [to date 23:59:59]
